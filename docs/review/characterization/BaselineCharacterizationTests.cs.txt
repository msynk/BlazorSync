using System.Runtime.CompilerServices;
using Bsync.Clocks;
using Bsync.Conflicts;
using Bsync.Protocol;
using Bsync.Server;
using Bsync.Storage;
using Bsync.Tests.TestSupport;
using Bsync.Transport;
using Xunit;

namespace Bsync.Tests;

// Desired-behaviour tests for the ten baseline scenarios, written against the API of commit 0083a02.
public sealed class BaselineCharacterizationTests
{
    [Fact]
    public async Task S01_PushDrainsAllBatches()
    {
        var server = new InMemorySyncServer<Note>();
        var engine = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(server),
            new HybridLogicalClock("a"), options: new SyncOptions<Note> { PushBatchSize = 2 });
        for (var i = 0; i < 5; i++) await engine.WriteAsync(new Note { Id = $"n{i}" });
        await engine.PushAsync();
        Assert.Equal(5, server.Snapshot().Count);
    }

    [Fact]
    public async Task S02_EditDuringAcknowledgementSurvives()
    {
        var server = new InMemorySyncServer<Note>();
        var inner = new InMemoryLocalStore<Note>();
        var store = new HookStore(inner);
        var transport = new HookTransport(new InProcessTransport<Note>(server));
        var engine = new SyncEngine<Note>(store, transport, new HybridLogicalClock("a"));
        await engine.WriteAsync(new Note { Id = "n1", Title = "v1" });
        transport.AfterPush = () => store.OnGet = id =>
        {
            store.OnGet = null;
            var rec = inner.GetAsync(id).Result!;
            rec.Current.Title = "v2";
            rec.Current.UpdatedAt = new HlcTimestamp(rec.Current.UpdatedAt.WallTime + 1, 0, "a");
            inner.UpsertAsync(rec with { IsDirty = true }).Wait();
        };
        await engine.PushAsync();
        var after = await engine.GetAsync("n1");
        Assert.Equal("v2", after!.Current.Title);
        Assert.True(after.IsDirty);
    }

    [Fact]
    public async Task S03_EditDuringPullSurvives()
    {
        var server = new InMemorySyncServer<Note>();
        var other = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(server), new HybridLogicalClock("b"));
        await other.WriteAsync(new Note { Id = "n1", Title = "base" });
        await other.SyncAsync();

        var inner = new InMemoryLocalStore<Note>();
        var store = new HookStore(inner);
        var transport = new HookTransport(new InProcessTransport<Note>(server));
        var engine = new SyncEngine<Note>(store, transport, new HybridLogicalClock("a"));
        await engine.PullAsync();

        await other.WriteAsync(new Note { Id = "n1", Title = "remote" });
        await other.SyncAsync();

        transport.AfterPull = () => store.OnGet = id =>
        {
            store.OnGet = null;
            var rec = inner.GetAsync(id).Result!;
            rec.Current.Title = "local";
            inner.UpsertAsync(rec with { IsDirty = true }).Wait();
        };
        await engine.PullAsync();
        var after = await engine.GetAsync("n1");
        Assert.Equal("local", after!.Current.Title);
        Assert.True(after.IsDirty);
    }

    [Fact]
    public async Task S04_LostPushResponseRetryDoesNotRepeatMerge()
    {
        var server = new InMemorySyncServer<Note>();
        var transport = new HookTransport(new InProcessTransport<Note>(server)) { LoseNextPushResponse = true };
        var merger = new DelegateConflictHandler<Note>(c =>
        {
            var m = c.Fork;
            m.Body = $"{c.RealMaster.Body}|{c.Fork.Body}";
            return ConflictResolution<Note>.Resolve(m);
        });
        var engine = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), transport, new HybridLogicalClock("a"), merger);
        await engine.WriteAsync(new Note { Id = "n1", Body = "one" });
        await Assert.ThrowsAsync<IOException>(() => engine.PushAsync());
        await engine.PushAsync();
        Assert.Equal("one", server.Snapshot()[0].Body);
    }

    [Fact]
    public async Task S05_FarFutureClientTimestampDoesNotPoisonServer()
    {
        var server = new InMemorySyncServer<Note>();
        var far = DateTimeOffset.UtcNow.AddYears(100).ToUnixTimeMilliseconds();
        var farClock = new HybridLogicalClock("evil", new ManualClock(far));
        var evil = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(server), farClock);
        var good = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(server), new HybridLogicalClock("good"));
        await evil.WriteAsync(new Note { Id = "evil" });
        try { await evil.SyncAsync(); } catch { /* rejection is acceptable */ }
        await good.WriteAsync(new Note { Id = "good" });
        await good.PushAsync();
        var stored = server.Snapshot().Single(n => n.Id == "good");
        Assert.True(stored.UpdatedAt.WallTime < far);
    }

    [Fact]
    public void S06_ClockCounterOverflowDoesNotBreakOrdering()
    {
        var clock = new HybridLogicalClock("a", new ManualClock(1_000));
        var remote = new HlcTimestamp(1_000, int.MaxValue, "b");
        HlcTimestamp result = default;
        var ex = Record.Exception(() => result = clock.Update(remote));
        Assert.True(ex is ArgumentException || result > remote);
    }

    [Fact]
    public void S07_EncodedOrderMatchesNumericOrder()
    {
        var ex = Record.Exception(() =>
        {
            var a = new HlcTimestamp(1_000, 999_999, "n");
            var b = new HlcTimestamp(1_000, 1_000_000, "n");
            Assert.True(a < b);
            Assert.True(string.CompareOrdinal(a.Encode(), b.Encode()) < 0);
        });
        Assert.True(ex is null or ArgumentException, ex?.Message);
    }

    [Fact]
    public async Task S08_RestartedClockDoesNotReuseTimestamp()
    {
        var physical = new ManualClock(1_000);
        var store = new InMemoryLocalStore<Note>();
        var server = new InMemorySyncServer<Note>();
        var first = new SyncEngine<Note>(store, new InProcessTransport<Note>(server), new HybridLogicalClock("a", physical));
        await first.WriteAsync(new Note { Id = "n1" });
        var restarted = new SyncEngine<Note>(store, new InProcessTransport<Note>(server), new HybridLogicalClock("a", physical));
        await restarted.WriteAsync(new Note { Id = "n2" });
        var t1 = (await restarted.GetAsync("n1"))!.Current.UpdatedAt;
        var t2 = (await restarted.GetAsync("n2"))!.Current.UpdatedAt;
        Assert.True(t2 > t1);
    }

    [Fact]
    public void S09_ZeroPushBatchSizeIsRejected()
    {
        Assert.ThrowsAny<ArgumentException>(() => new SyncEngine<Note>(
            new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(new InMemorySyncServer<Note>()),
            new HybridLogicalClock("a"), options: new SyncOptions<Note> { PushBatchSize = 0 }));
    }

    [Fact]
    public async Task S10_LastWriteWinsIsIndependentOfUploadOrder()
    {
        Assert.Equal(await RunLww(aFirst: true), await RunLww(aFirst: false));

        static async Task<string> RunLww(bool aFirst)
        {
            var server = new InMemorySyncServer<Note>();
            var lww = new LastWriteWinsConflictHandler<Note>();
            var past = DateTimeOffset.UtcNow.ToUnixTimeMilliseconds() - 60_000;
            var a = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(server), new HybridLogicalClock("a", new ManualClock(past)), lww);
            var b = new SyncEngine<Note>(new InMemoryLocalStore<Note>(), new InProcessTransport<Note>(server), new HybridLogicalClock("b", new ManualClock(past)), lww);
            await a.WriteAsync(new Note { Id = "n1", Title = "base" });
            await a.SyncAsync();
            await b.SyncAsync();
            await a.WriteAsync(new Note { Id = "n1", Title = "A" });
            await b.WriteAsync(new Note { Id = "n1", Title = "B0" });
            await b.WriteAsync(new Note { Id = "n1", Title = "B" });
            Assert.True((await b.GetAsync("n1"))!.Current.UpdatedAt > (await a.GetAsync("n1"))!.Current.UpdatedAt);
            await Task.Delay(20); // server re-stamps accepted writes with its (later) physical clock
            if (aFirst) { await a.SyncAsync(); await b.SyncAsync(); }
            else { await b.SyncAsync(); await a.SyncAsync(); }
            return server.Snapshot()[0].Title;
        }
    }

    private sealed class HookStore(ILocalStore<Note> inner) : ILocalStore<Note>
    {
        // Runs between the engine's read and its write of the same record.
        public Action<string>? OnGet { get; set; }

        public Task<SyncRecord<Note>?> GetAsync(string id, CancellationToken cancellationToken = default) => inner.GetAsync(id, cancellationToken);

        public Task UpsertAsync(SyncRecord<Note> record, CancellationToken cancellationToken = default)
        {
            OnGet?.Invoke(record.Current.Id);
            return inner.UpsertAsync(record, cancellationToken);
        }
        public Task<IReadOnlyList<SyncRecord<Note>>> GetDirtyAsync(int batchSize, CancellationToken cancellationToken = default) => inner.GetDirtyAsync(batchSize, cancellationToken);
        public Task<IReadOnlyList<Note>> QueryAsync(bool includeDeleted = false, CancellationToken cancellationToken = default) => inner.QueryAsync(includeDeleted, cancellationToken);
        public Task<Checkpoint> GetCheckpointAsync(CancellationToken cancellationToken = default) => inner.GetCheckpointAsync(cancellationToken);
        public Task SetCheckpointAsync(Checkpoint checkpoint, CancellationToken cancellationToken = default) => inner.SetCheckpointAsync(checkpoint, cancellationToken);
    }

    private sealed class HookTransport(ISyncTransport<Note> inner) : ISyncTransport<Note>
    {
        public Action? AfterPush { get; set; }
        public Action? AfterPull { get; set; }
        public bool LoseNextPushResponse { get; set; }

        public async Task<PullResult<Note>> PullAsync(PullRequest request, CancellationToken cancellationToken = default)
        {
            var result = await inner.PullAsync(request, cancellationToken);
            AfterPull?.Invoke();
            AfterPull = null;
            return result;
        }

        public async Task<PushResult<Note>> PushAsync(PushRequest<Note> request, CancellationToken cancellationToken = default)
        {
            var result = await inner.PushAsync(request, cancellationToken);
            if (LoseNextPushResponse)
            {
                LoseNextPushResponse = false;
                throw new IOException("response lost");
            }

            AfterPush?.Invoke();
            AfterPush = null;
            return result;
        }

        public IAsyncEnumerable<StreamEvent<Note>> StreamAsync(Checkpoint since, CancellationToken cancellationToken = default) =>
            inner.StreamAsync(since, cancellationToken);
    }
}
